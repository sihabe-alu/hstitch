**Goal:** 사진 2~3장을 리사이즈 없이 가로로 이어붙이는 개인용 웹UI (mac mini 서버, Windows 브라우저 접속).

**Architecture:** Flask 앱 하나(`app.py`)에 합성 함수 `stitch()`와 라우트 2개. 프론트는 단일 `index.html`(빌드 없음). 메모리 내 처리, 저장 없음.

**Tech Stack:** Python 3, Flask, Pillow, pytest.

**Spec:** `docs/superpowers/specs/2026-09-26-hstitch-design.md`

## Global Constraints

- 위치: `~/hstitch/` (ComfyUI repo 밖)
- 의존성은 `flask`, `pillow`만 (테스트용 `pytest` 제외)
- 리사이즈·크롭 금지. 캔버스 = `(sum(w), max(h))`, 작은 사진은 세로 중앙, 여백 흰색
- PNG 기본. JPEG는 `quality=100, subsampling=0`. 첫 이미지 ICC 프로파일 유지
- `ImageOps.exif_transpose` 적용
- 이미지 개수 검증 없음(1장 그대로, 4장 이상도 합침). 디코드 실패만 400
- 인증 없음, 서버 저장 없음, tailscale IP 바인딩

## Review Focus

- EXIF 회전 JPEG(폰 사진): 회전 반영된 크기로 합성되어야 함 → Task 1 테스트
- 알파 있는 PNG: 투명 영역이 검정이 아니라 흰색이어야 함 → Task 1 테스트
- 팔레트/그레이스케일 모드(P, L): 에러 없이 합성 → Task 1 테스트
- 업로드 0개: 500이 아니라 400 → Task 2 테스트
- 알 수 없는 `format` 값: 에러 대신 PNG → Task 2 테스트

```python
import io

import pytest
from PIL import Image, UnidentifiedImageError

from app import stitch


def make(size, color, mode="RGB", fmt="PNG", **save_kw):
    buf = io.BytesIO()
    Image.new(mode, size, color).save(buf, fmt, **save_kw)
    buf.seek(0)
    return buf


def open_result(data):
    return Image.open(io.BytesIO(data))


def test_size_is_sum_of_widths_and_max_height():
    out = open_result(stitch([make((30, 20), "red"), make((10, 50), "blue")]))
    assert out.size == (40, 50)


def test_png_is_lossless_and_centered_with_white_padding():
    out = open_result(stitch([make((30, 20), (255, 0, 0)), make((10, 50), (0, 0, 255))])).convert("RGB")
    assert out.getpixel((0, 15)) == (255, 0, 0)      # red starts at y=(50-20)//2=15
    assert out.getpixel((29, 34)) == (255, 0, 0)
    assert out.getpixel((0, 0)) == (255, 255, 255)   # padding above red
    assert out.getpixel((0, 49)) == (255, 255, 255)  # padding below red
    assert out.getpixel((30, 0)) == (0, 0, 255)      # blue fills full height


def test_jpeg_output():
    out = open_result(stitch([make((8, 8), "red"), make((8, 8), "blue")], "jpeg"))
    assert out.format == "JPEG"


def test_exif_rotation_is_applied():
    exif = Image.Exif()
    exif[0x0112] = 6  # rotate 90 CW when displayed
    rotated = make((40, 10), "red", fmt="JPEG", exif=exif.tobytes())
    out = open_result(stitch([rotated, make((5, 5), "blue")]))
    assert out.size == (10 + 5, 40)


def test_transparent_png_padding_is_white_not_black():
    clear = make((10, 10), (0, 0, 0, 0), mode="RGBA")
    out = open_result(stitch([clear, make((10, 10), "blue")])).convert("RGB")
    assert out.getpixel((5, 5)) == (255, 255, 255)


@pytest.mark.parametrize("mode,color", [("P", 3), ("L", 128)])
def test_non_rgb_modes(mode, color):
    out = open_result(stitch([make((6, 6), color, mode=mode), make((6, 9), "blue")]))
    assert out.size == (12, 9)


def test_corrupt_file_raises():
    with pytest.raises(UnidentifiedImageError):
        stitch([io.BytesIO(b"not an image")])


def test_empty_raises_value_error():
    with pytest.raises(ValueError):
        stitch([])
```

```python
import io

from PIL import Image, ImageOps


def stitch(files, fmt="png"):
    if not files:
        raise ValueError("no images")
    images = [ImageOps.exif_transpose(Image.open(f)) for f in files]
    width = sum(im.width for im in images)
    height = max(im.height for im in images)
    canvas = Image.new("RGB", (width, height), "white")
    x = 0
    for im in images:
        rgba = im.convert("RGBA")
        canvas.paste(rgba, (x, (height - im.height) // 2), rgba)
        x += im.width
    icc = images[0].info.get("icc_profile")
    out = io.BytesIO()
    if fmt == "jpeg":
        canvas.save(out, "JPEG", quality=100, subsampling=0, icc_profile=icc)
    else:
        canvas.save(out, "PNG", icc_profile=icc)
    return out.getvalue()
```

```python
import io

from PIL import Image

from app import app


def png(size, color):
    buf = io.BytesIO()
    Image.new("RGB", size, color).save(buf, "PNG")
    buf.seek(0)
    return buf


def post(client, files, **data):
    return client.post(
        "/stitch",
        data={"images": [(f, "x.png") for f in files], **data},
        content_type="multipart/form-data",
    )


def test_stitch_route_returns_png_by_default():
    r = post(app.test_client(), [png((10, 20), "red"), png((10, 30), "blue")])
    assert r.status_code == 200
    assert r.mimetype == "image/png"
    assert Image.open(io.BytesIO(r.data)).size == (20, 30)


def test_jpeg_format():
    r = post(app.test_client(), [png((8, 8), "red")], format="jpeg")
    assert r.mimetype == "image/jpeg"


def test_unknown_format_falls_back_to_png():
    r = post(app.test_client(), [png((8, 8), "red")], format="bmp")
    assert r.status_code == 200
    assert r.mimetype == "image/png"


def test_corrupt_upload_is_400():
    r = post(app.test_client(), [io.BytesIO(b"junk")])
    assert r.status_code == 400


def test_no_files_is_400():
    r = post(app.test_client(), [])
    assert r.status_code == 400
```

```python
@app.get("/")
def index():
    return send_file("index.html")


@app.post("/stitch")
def stitch_route():
    fmt = "jpeg" if request.form.get("format") == "jpeg" else "png"
    try:
        data = stitch([f.stream for f in request.files.getlist("images")], fmt)
    except (ValueError, UnidentifiedImageError) as e:
        return Response(str(e), status=400, mimetype="text/plain")
    return Response(data, mimetype=f"image/{fmt}")
```

```html
const $ = id => document.getElementById(id);
let files = [];

function render() {
  $('thumbs').replaceChildren(...files.map((f, i) => {
    const d = document.createElement('div'); d.className = 't';
    const img = document.createElement('img'); img.src = URL.createObjectURL(f);
    const l = document.createElement('button'); l.textContent = '←'; l.disabled = i === 0;
    l.onclick = () => { [files[i-1], files[i]] = [files[i], files[i-1]]; render(); };
    const r = document.createElement('button'); r.textContent = '→'; r.disabled = i === files.length - 1;
    r.onclick = () => { [files[i+1], files[i]] = [files[i], files[i+1]]; render(); };
    const x = document.createElement('button'); x.textContent = '×';
    x.onclick = () => { files.splice(i, 1); render(); };
    d.append(img, l, r, x);
    return d;

function add(list) { files.push(...[...list].filter(f => f.type.startsWith('image/'))); render(); }

$('file').onchange = e => { add(e.target.files); e.target.value = ''; };
```
